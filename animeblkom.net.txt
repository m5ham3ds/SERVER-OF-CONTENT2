الملف الجديد: AnimeBlkom.kt

```kotlin
package eu.kanade.tachiyomi.animeextension.ar.animeblkom

import android.util.Log
import androidx.preference.ListPreference
import androidx.preference.PreferenceScreen
import eu.kanade.tachiyomi.animesource.ConfigurableAnimeSource
import eu.kanade.tachiyomi.animesource.model.AnimeFilter
import eu.kanade.tachiyomi.animesource.model.AnimeFilterList
import eu.kanade.tachiyomi.animesource.model.SAnime
import eu.kanade.tachiyomi.animesource.model.SEpisode
import eu.kanade.tachiyomi.animesource.model.Video
import eu.kanade.tachiyomi.animesource.online.ParsedAnimeHttpSource
import eu.kanade.tachiyomi.network.GET
import eu.kanade.tachiyomi.network.await
import eu.kanade.tachiyomi.network.awaitSuccess
import keiyoushi.utils.getPreferencesLazy
import keiyoushi.utils.useAsJsoup
import okhttp3.Request
import okhttp3.Response
import org.jsoup.nodes.Document
import org.jsoup.nodes.Element
import kotlin.math.ceil

class AnimeBlkom : ParsedAnimeHttpSource(), ConfigurableAnimeSource {

    override val name = "AnimeBlkom"
    override val baseUrl = "https://animeblkom.net"
    override val lang = "ar"
    override val supportsLatest = true

    private val preferences by getPreferencesLazy()

    // =============================== Popular ===============================
    // نستخدم صفحة قائمة الأنمي: /anime-list
    // تدعم pagination عبر ?page=X
    override fun popularAnimeSelector(): String = "div.content"

    override fun popularAnimeNextPageSelector(): String = "ul.pagination li.page-item:not(.disabled) a[rel=next]"

    override fun popularAnimeRequest(page: Int): Request {
        val url = if (page == 1) {
            "$baseUrl/anime-list"
        } else {
            "$baseUrl/anime-list?page=$page"
        }
        return GET(url, headers)
    }

    override fun popularAnimeFromElement(element: Element): SAnime {
        val anime = SAnime.create()
        // الرابط: a داخل .poster
        val linkElement = element.selectFirst("div.poster a")
        anime.setUrlWithoutDomain(linkElement?.attr("href") ?: "")
        // العنوان: داخل .name a
        anime.title = element.selectFirst("div.name a")?.text()?.trim() ?: "بدون عنوان"
        // الصورة: img داخل .poster (data-original أو src)
        val img = element.selectFirst("div.poster img")
        anime.thumbnail_url = img?.attr("data-original")?.ifEmpty { img.attr("src") } ?: ""
        return anime
    }

    // =============================== Latest ===============================
    // نستخدم نفس القائمة (يمكن اعتبار أنها أحدث الإضافات)
    override fun latestUpdatesSelector(): String = popularAnimeSelector()
    override fun latestUpdatesNextPageSelector(): String = popularAnimeNextPageSelector()
    override fun latestUpdatesRequest(page: Int): Request = popularAnimeRequest(page)
    override fun latestUpdatesFromElement(element: Element): SAnime = popularAnimeFromElement(element)

    // =============================== Search ===============================
    override fun searchAnimeSelector(): String = popularAnimeSelector()

    override fun searchAnimeNextPageSelector(): String = popularAnimeNextPageSelector()

    override fun searchAnimeRequest(page: Int, query: String, filters: AnimeFilterList): Request {
        val url = if (query.isNotBlank()) {
            // البحث: /search?query=...
            "$baseUrl/search?query=${query.replace(" ", "+")}&page=$page"
        } else {
            // إذا لم يوجد بحث، نستخدم القائمة مع فلتر إن وجد
            var finalUrl = "$baseUrl/anime-list"
            // تطبيق الفلتر إن وجد (نوع الأنمي مثلاً)
            var type = ""
            for (filter in filters) {
                if (filter is TypeFilter && filter.state > 0) {
                    type = filter.values[filter.state]
                    break
                }
            }
            if (type.isNotEmpty()) {
                finalUrl += when (type) {
                    "مسلسلات" -> "/anime-list"
                    "أفلام" -> "/movie-list"
                    "أوفا" -> "/ova-list"
                    "أونا" -> "/ona-list"
                    "حلقات خاصة" -> "/special-list"
                    else -> "/anime-list"
                }
            }
            if (page > 1) {
                finalUrl += if (finalUrl.contains("?")) "&page=$page" else "?page=$page"
            }
            return GET(finalUrl, headers)
        }
        return GET(url, headers)
    }

    override fun searchAnimeFromElement(element: Element): SAnime = popularAnimeFromElement(element)

    // =============================== Filters ===============================
    override fun getFilterList(): AnimeFilterList {
        return AnimeFilterList(
            TypeFilter("النوع", arrayOf("الكل", "مسلسلات", "أفلام", "أوفا", "أونا", "حلقات خاصة"))
        )
    }

    private class TypeFilter(name: String, vals: Array<String>) :
        AnimeFilter.Select<String>(name, vals)

    // =============================== Episodes ===============================
    override fun episodeListSelector(): String = "ul.episodes-links li.episode-link"

    override fun episodeFromElement(element: Element): SEpisode {
        val episode = SEpisode.create()
        val link = element.selectFirst("a")?.attr("href") ?: ""
        episode.setUrlWithoutDomain(link)
        // اسم الحلقة: نأخذ النص (مثلاً "الحلقة 1")
        val text = element.text().trim()
        episode.name = text
        // استخراج رقم الحلقة
        val number = text.filter { it.isDigit() }.toFloatOrNull() ?: 0f
        episode.episode_number = number
        return episode
    }

    // =============================== Anime Details ===============================
    override fun animeDetailsParse(document: Document): SAnime {
        val anime = SAnime.create()
        // الصورة
        anime.thumbnail_url = document.selectFirst("div.poster img")?.attr("data-original")?.ifEmpty { 
            document.selectFirst("div.poster img")?.attr("src") 
        } ?: ""
        // العنوان
        anime.title = document.selectFirst("div.name h1")?.text()?.trim() ?: "بدون عنوان"
        // الوصف
        anime.description = document.selectFirst("div.story")?.text()?.trim() ?: ""
        // التصنيفات (genres)
        val genres = document.select("p.genres a").eachText()
        anime.genre = genres.joinToString(", ")
        // الحالة: نبحث في info-boxes
        val statusElement = document.selectFirst("div.info-table div:has(span.head:contains(حالة الأنمي)) span.info")
        val statusText = statusElement?.text()?.trim() ?: ""
        anime.status = when {
            statusText.contains("منتهي", ignoreCase = true) -> SAnime.COMPLETED
            statusText.contains("مستمر", ignoreCase = true) -> SAnime.ONGOING
            statusText.contains("قادم", ignoreCase = true) -> SAnime.UPCOMING
            else -> SAnime.UNKNOWN
        }
        // معلومات إضافية يمكن استخراجها: عدد الحلقات، الاستوديو، إلخ.
        // لا نضيفها حالياً
        return anime
    }

    // =============================== Videos ===============================
    override suspend fun getVideoList(episode: SEpisode): List<Video> {
        // صفحة المشاهدة: /watch/...
        val document = client.newCall(GET(baseUrl + episode.url, headers))
            .await()
            .useAsJsoup()

        val videoList = mutableListOf<Video>()
        // السيرفرات: توجد داخل div.servers عناصر span.server مع a[data-src]
        val serverElements = document.select("div.servers span.server a[data-src]")
        for (a in serverElements) {
            val src = a.attr("data-src")
            if (src.isNotBlank()) {
                // نأخذ اسم السيرفر من النص
                val serverName = a.text().trim()
                // نحدد الجودة (غالباً غير موجودة، نضع "متعدد")
                val quality = "متعدد"
                videoList.add(Video(src, "$serverName - $quality", src))
            }
        }

        // إذا لم نجد سيرفرات، قد نجد روابط تحميل مباشرة في #download
        if (videoList.isEmpty()) {
            val downloadLinks = document.select("#download a[href]")
            for (link in downloadLinks) {
                val url = link.attr("href")
                if (url.isNotBlank() && !url.contains("javascript:")) {
                    val quality = link.selectFirst("small")?.text()?.trim() ?: "تحميل"
                    videoList.add(Video(url, "Download - $quality", url))
                }
            }
        }

        return videoList
    }

    override fun videoListSelector(): String = "div.servers span.server a[data-src]"

    override fun videoFromElement(element: Element): Video {
        throw UnsupportedOperationException("يتم التعامل مع الفيديوهات مباشرة في getVideoList")
    }

    override fun videoUrlParse(document: Document): String {
        throw UnsupportedOperationException("لا حاجة لهذه الدالة")
    }

    // ترتيب الفيديوهات (اختياري)
    override fun List<Video>.sort(): List<Video> {
        // يمكننا ترتيب حسب الجودة إذا أردنا، لكننا سنكتفي بالترتيب الحالي
        return this
    }

    // =============================== Preferences ===============================
    override fun setupPreferenceScreen(screen: PreferenceScreen) {
        // يمكن إضافة تفضيلات لاحقاً
    }
}
```

شرح تفصيلي للملف وأسلوب عمله

1. استخراج القوائم (Popular, Latest, Search)

· نستخدم صفحة قائمة الأنمي (/anime-list) التي تدعم التصفح (pagination) بإضافة ?page=X.
· المحددات (Selectors) المستخدمة:
  · div.content لكل بطاقة أنمي (نستخدم div.content لأن كل أنمي موضوع داخل div.content).
  · div.poster a للرابط.
  · div.name a للعنوان.
  · div.poster img للصورة (نأخذ data-original أولاً، ثم src).

2. التصفية (Filters)

· أضفنا فلتراً للنوع (مسلسلات، أفلام، أوفا، أونا، حلقات خاصة) لتوجيه المستخدم إلى القوائم المناسبة.
· عند اختيار نوع، يتم بناء الرابط المناسب (/anime-list أو /movie-list، إلخ).

3. جلب الحلقات (Episodes)

· في صفحة تفاصيل الأنمي، تظهر الحلقات في قائمة ul.episodes-links داخل li.episode-link.
· نستخرج الرابط من <a> ونأخذ النص كاسم الحلقة (مثل "الحلقة 1").
· نستخرج رقم الحلقة من النص باستخدام التصفية على الأرقام.

4. تفاصيل الأنمي (Anime Details)

· نستخرج البيانات من div.anime-info-section:
  · الصورة: div.poster img (نأخذ data-original أو src).
  · العنوان: div.name h1.
  · الوصف: div.story.
  · التصنيفات (genre): ندمج جميع p.genres a في سلسلة مفصولة بفواصل.
  · الحالة: نبحث في div.info-table عن span.head الذي يحوي "حالة الأنمي" ثم نأخذ span.info المجاور ونحدد الحالة (منتهي، مستمر، قادم).

5. جلب روابط التشغيل (Videos)

· في صفحة الحلقة (/watch/...)، نجد قائمة السيرفرات داخل div.servers.
· كل عنصر span.server يحتوي على <a data-src="..."> الذي يحوي رابط الـ iframe.
· نستخرج هذه الروابط ونضيفها كفيديوهات.
· إذا لم نجد سيرفرات، نبحث عن روابط تحميل مباشرة في #download (لكننا نفضل السيرفرات أولاً).

6. الإعدادات (Preferences)

· لم نضف أي تفضيلات حالياً، لكن يمكن إضافة خيارات مثل الجودة المفضلة في المستقبل.

7. التعامل مع pagination في البحث

· عند البحث (searchAnimeRequest) نضيف ?query=...&page=X للبحث، وعند عدم وجود بحث نستخدم القائمة مع إمكانية تطبيق التصنيف.

8. ملاحظات مهمة

· بعض الصفحات قد تحتوي على روابط تحميل مباشرة (مثل #download)، وقد نضيفها كفيديوهات إضافية.
· قد تحتاج بعض السيرفرات إلى معالجة خاصة (مثل استخراج الفيديو من iframe بعد تحميل الصفحة)، لكننا في هذا الكود نمرر الرابط مباشرة ليعمل التطبيق على عرضه داخل iframe.

· عرض قوائم الأنمي (الأكثر شهرة، الأحدث، البحث).

· جلب تفاصيل الأنمي.
· جلب قائمة الحلقات.
· جلب روابط تشغيل الفيديو من سيرفرات متعددة.
