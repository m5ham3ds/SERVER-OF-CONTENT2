    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const header = document.getElementById('mainHeader');
            const bottomNav = document.getElementById('bottom-nav');

            // --- Desktop Search Elements ---
            const desktopSearchInput = document.getElementById('desktopSearchInput');
            const desktopSearchButton = document.getElementById('desktopSearchButton');
            const desktopSuggestionsList = document.getElementById('desktopSuggestionsList');

            // --- Mobile Search Elements ---
            const mobileSearchInput = document.getElementById('mobileSearchInput');
            const mobilePerformSearchButton = document.getElementById('mobilePerformSearchButton');
            const mobileSuggestionsList = document.getElementById('mobileSuggestionsList');

            // --- Other Mobile UI Elements (Moved to top for scope) ---
            const mobileSearchBtn = document.getElementById('mobileSearchBtn');
            const mobileSearchContainer = document.getElementById('mobileSearchContainer');
            const closeMobileSearch = document.getElementById('closeMobileSearch'); // This was causing the error

            const moreMenuBtn = document.getElementById('more-menu-btn');
            const moreMenuPanel = document.getElementById('more-menu-panel');
            const closeMoreMenuBtn = document.getElementById('close-more-menu');
            const menuOverlay = document.getElementById('menu-overlay');

            const authModal = document.getElementById('auth-modal-overlay');
            const openAuthBtnDesktop = document.getElementById('open-auth-modal-desktop');
            const openAuthBtnMobile = document.getElementById('open-auth-modal-mobile');
            const closeAuthBtn = document.getElementById('close-auth-modal');


            let debounceTimer;

            // Function to perform search (redirects to search results page)
            const performSearch = (inputElement) => {
                const query = inputElement.value.trim();
                if (query) {
                    window.location.href = `https://www.stardima.com/search?query=${encodeURIComponent(query)}`;
                }
            };

            // Function to fetch and display suggestions
            const fetchSuggestions = (inputElement, suggestionsListElement) => {
                clearTimeout(debounceTimer);
                const query = inputElement.value.trim();

                if (query.length > 0) {
                    debounceTimer = setTimeout(() => {
                        fetch(`https://www.stardima.com/search?query=${encodeURIComponent(query)}`, {
                                headers: {
                                    'X-Requested-With': 'XMLHttpRequest'
                                }
                            })
                            .then(response => response.json())
                            .then(data => {
                                suggestionsListElement.innerHTML = ''; // Clear previous
                                if (data.length > 0) {
                                    data.forEach(video => {
                                        const suggestionItem = document.createElement('a');
                                        suggestionItem.href = video.url;
                                        suggestionItem.classList.add('flex',
                                            'items-center', 'p-2',
                                            'hover:bg-gray-700',
                                            'transition-colors', 'rounded-md');
                                        suggestionItem.innerHTML = `
                                            <img src="${video.image}" alt="${video.title}" class="w-10 h-10 object-cover rounded mr-2">
                                            <div class="flex-1">
                                                <p class="text-white text-sm font-medium">${video.title}</p>
                                                <p class="text-gray-400 text-xs">${video.year} - ${video.quality}</p>
                                            </div>
                                        `;
                                        suggestionsListElement.appendChild(suggestionItem);
                                    });
                                    suggestionsListElement.classList.remove('hidden');
                                } else {
                                    suggestionsListElement.classList.add('hidden');
                                }
                            })
                            .catch(error => {
                                console.error('Error fetching search suggestions:', error);
                                suggestionsListElement.classList.add('hidden');
                            });
                    }, 300);
                } else {
                    suggestionsListElement.classList.add('hidden');
                }
            };

            // --- Desktop Search Event Listeners ---
            if (desktopSearchInput) {
                desktopSearchInput.addEventListener('input', () => fetchSuggestions(desktopSearchInput,
                    desktopSuggestionsList));
                desktopSearchButton.addEventListener('click', () => performSearch(desktopSearchInput));
                desktopSearchInput.addEventListener('keypress', (e) => {
                    if (e.key === 'Enter') {
                        performSearch(desktopSearchInput);
                    }
                });
                document.addEventListener('click', (event) => {
                    if (!desktopSuggestionsList.contains(event.target) && event.target !==
                        desktopSearchInput) {
                        desktopSuggestionsList.classList.add('hidden');
                    }
                });
            }


            // --- Mobile Search Event Listeners ---
            if (mobileSearchInput) {
                mobileSearchInput.addEventListener('input', () => fetchSuggestions(mobileSearchInput,
                    mobileSuggestionsList));
                mobilePerformSearchButton.addEventListener('click', () => performSearch(mobileSearchInput));
                mobileSearchInput.addEventListener('keypress', (e) => {
                    if (e.key === 'Enter') {
                        performSearch(mobileSearchInput);
                    }
                });
                // Special handling for mobile to hide suggestions when clicking outside or closing overlay
                document.addEventListener('click', (event) => {
                    if (!mobileSuggestionsList.contains(event.target) && event.target !==
                        mobileSearchInput && !mobileSearchContainer.contains(event.target)) {
                        mobileSuggestionsList.classList.add('hidden');
                    }
                });
                // Ensure suggestions are hidden when mobile search is closed
                // This listener must come *after* closeMobileSearch is defined.
                closeMobileSearch.addEventListener('click', () => {
                    mobileSuggestionsList.classList.add('hidden');
                    mobileSearchInput.value = ''; // Clear input on close
                });
            }


            // --- Header & Bottom Nav Behavior on Scroll ---
            let lastScrollY = window.scrollY;
            const handleScroll = () => {
                // For Desktop Header
                if (window.innerWidth >= 1024) {
                    if (window.scrollY > 50) {
                        header.classList.add('bg-body/80', 'backdrop-blur');
                    } else {
                        header.classList.remove('bg-body/80', 'backdrop-blur');
                    }
                }
                // For Mobile Bottom Nav
                if (window.innerWidth < 1024) {
                    if (window.scrollY > lastScrollY) {
                        // Scrolling Down
                        bottomNav.style.transform = 'translateY(100%)';
                    } else {
                        // Scrolling Up
                        bottomNav.style.transform = 'translateY(0)';
                    }
                }
                lastScrollY = window.scrollY;
            };
            window.addEventListener('scroll', handleScroll, {
                passive: true
            });
            handleScroll(); // Initial check

            // --- "More" Menu Panel Logic ---
            const openMoreMenu = () => {
                moreMenuPanel.classList.remove('translate-y-full');
                menuOverlay.classList.remove('hidden');
                document.body.style.overflow = 'hidden';
            };

            const closeMoreMenu = () => {
                moreMenuPanel.classList.add('translate-y-full');
                menuOverlay.classList.add('hidden');
                document.body.style.overflow = '';
            };

            moreMenuBtn?.addEventListener('click', openMoreMenu);
            closeMoreMenuBtn?.addEventListener('click', closeMoreMenu);
            menuOverlay?.addEventListener('click', closeMoreMenu);

            // --- Mobile Search Overlay Logic ---
            const openMobileSearch = () => {
                mobileSearchContainer.classList.remove('-translate-y-full');
                document.body.style.overflow = 'hidden';
                mobileSearchInput.focus(); // Focus the input when opened
            };

            const closeMobileSearchFunc = () => {
                mobileSearchContainer.classList.add('-translate-y-full');
                document.body.style.overflow = '';
                mobileSuggestionsList.classList.add('hidden'); // Ensure hidden on close
                mobileSearchInput.value = ''; // Clear input on close
            };

            mobileSearchBtn?.addEventListener('click', openMobileSearch);
            closeMobileSearch?.addEventListener('click', closeMobileSearchFunc);


            // --- Auth Modal Logic ---
            const openModal = () => {
                if (authModal) {
                    authModal.classList.remove('hidden');
                    closeMoreMenu(); // Close the 'more' panel if open
                }
            };

            const closeModal = () => {
                if (authModal) authModal.classList.add('hidden');
            };

            if (openAuthBtnDesktop) openAuthBtnDesktop.addEventListener('click', openModal);
            if (openAuthBtnMobile) openAuthBtnMobile.addEventListener('click', openModal);
            if (closeAuthBtn) closeAuthBtn.addEventListener('click', closeModal);

            if (authModal) {
                authModal.addEventListener('click', (e) => {
                    if (e.target === authModal) closeModal();
                });
            }
        });
    </script>
